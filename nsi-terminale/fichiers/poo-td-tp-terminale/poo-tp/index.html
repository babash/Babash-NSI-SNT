Dans une méthode, la carte elle-même est self et l'autre carte est autre : on compare self.valeur et autre.valeur.

Une comparaison vaut déjà True ou False : return self.valeur == autre.valeur suffit.

class Carte:
    def __init__(self, valeur, couleur, enseigne):
        self.valeur = valeur
        self.couleur = couleur
        self.enseigne = enseigne

    def egalite(self, autre):
        """Renvoie True si self et autre ont la même valeur, False sinon."""
        return self.valeur == autre.valeur

    def estSuperieurA(self, autre):
        """Renvoie True si self a une valeur plus grande que autre."""
        return self.valeur > autre.valeur

    def estInferieurA(self, autre):
        """Renvoie True si self a une valeur plus petite que autre."""
        return self.valeur < autre.valeur

Une méthode s'écrit comme une fonction, dans la classe, avec self en premier paramètre. On l'appelle sur un objet : roi.estSuperieurA(sept).

        self.cartes = []    # la liste des cartes du jeu
        for enseigne in ['carreau', 'coeur', 'trèfle', 'pique']:
            if enseigne == 'carreau' or enseigne == 'coeur':

    def distribuerCarte(self):
        """Enlève la première carte du jeu et la renvoie."""
        return self.cartes.pop(0)

import random
random.seed(2046)   # hasard fixé : le même mélange pour tout le monde


class Carte:
    def __init__(self, valeur, couleur, enseigne):
        self.valeur = valeur
        self.couleur = couleur
        self.enseigne = enseigne


class JeuCartes:
    def __init__(self):
        self.cartes = []    # la liste des cartes du jeu
        for enseigne in ['carreau', 'coeur', 'trèfle', 'pique']:
            if enseigne == 'carreau' or enseigne == 'coeur':
                couleur = 'rouge'
            else:
                couleur = 'noir'
            for valeur in range(2, 15):
                self.cartes.append(Carte(valeur, couleur, enseigne))

    def melanger(self):
        random.shuffle(self.cartes)

    def distribuerCarte(self):
        return self.cartes.pop(0)


# Programme principal
jeu = JeuCartes()
jeu.melanger()
while len(jeu.cartes) > 0:
    carte = jeu.distribuerCarte()
    print(carte.valeur, carte.enseigne, carte.couleur)

On crée l'objet, on appelle ses méthodes, et la boucle s'arrête quand le jeu est vide. Sans random.seed, le mélange change à chaque lancement.

Mémo

Une classe décrit des objets : son constructeur __init__ crée leurs attributs (self.valeur), ses méthodes agissent sur eux (roi.estSuperieurA(sept)).

Une instance se crée en appelant la classe : c1 = Carte(7, 'rouge', 'coeur').
