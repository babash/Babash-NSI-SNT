\section*{Exercices d'entrainement}
%\textbf{\huge{ Exercices d'entraînement - Chapitres 1 et 2}}\\[1cm]
\begin{Exercise}[title = {Comprendre les bases 2, 10 et 16}]
    \Question Écrire le nombre $15_{(10)}$ en hexadécimal et binaire
    \Question Écrire le nombre $10000_{(2)}$ en décimal et hexadécimal
    \Question Écrire le nombre $1F_{(16)}$ en décimal
\end{Exercise}
\vspace{-0.7cm}
\begin{Exercise}[title = Conversion de la base 16 à la base 10]
    \Question Convertir en décimal le nombre $FF_{(16)}$
    \Question Convertir en décimal le nombre $A38_{(16)}$
    \Question Convertir en décimal le nombre $B38_{(16)}$
\end{Exercise}
\vspace{-0.7cm}
\begin{Exercise}[title = Conversion de la base 2 à la base 10]
    \Question Convertir en décimal le nombre $1111_{(2)}$
    \Question Convertir en décimal le nombre $10000_{(2)}$
    \Question Convertir en décimal le nombre $10010_{(2)}$
    \Question Convertir en décimal le nombre $1111 1111_{(2)}$
\end{Exercise}
\vspace{-0.7cm}
\begin{Exercise} [title = Conversion de la base 10 à la base 2]
    \Question Convertir en binaire le nombre $80_{(10)}$
    \Question Convertir en binaire le nombre $255_{(10)}$
    \Question Convertir en binaire le nombre $256_{(10)}$
\end{Exercise}
\vspace{-0.7cm}
\begin{Exercise}[title = Conversion de la base 10 à la base 16]
    \Question Convertir en hexadécimal le nombre $129_{(10)}$
    \Question Convertir en hexadécimal le nombre $256_{(10)}$
    \Question Convertir en hexadécimal le nombre $257_{(10)}$
    \Question Convertir en hexadécimal le nombre $1920_{(10)}$
\end{Exercise}
\vspace{-0.7cm}
\begin{Exercise}[title = Conversion de la base 2 à la base 16]
    \Question Convertir en hexadécimal le nombre $1001_{(2)}$
    \Question Convertir en hexadécimal le nombre $1011_{(2)}$
    \Question Convertir en hexadécimal le nombre $1001~1011_{(2)}$
    \Question Convertir en hexadécimal le nombre $1111_{(2)}$
    \Question Convertir en hexadécimal le nombre $1111~ 1111_{(2)}$ 
\end{Exercise}
\vspace{-0.7cm}
\begin{Exercise}[title = Bonus]
    \Question Convertir votre date de naissance en hexadécimal ou en décimal et votre voisin doit faire la conversion !
\end{Exercise}
