def cesar_general(mot, n):
    alphabet = "abcdefghijklmnopqrstuvwxyz"
    resultat = ""
    for lettre in mot:
        if lettre in alphabet:
            indice = (alphabet.index(lettre) + n) % 26
            resultat = resultat + alphabet[indice]
        else:
            resultat = resultat + lettre
    return resultat

def vigenere_liste(mot, decalages):
    alphabet = "abcdefghijklmnopqrstuvwxyz"
    resultat = ""
    i = 0
    for lettre in mot:
        indice = (alphabet.index(lettre) + decalages[i]) % 26
        resultat = resultat + alphabet[indice]
        i = i + 1
        if i == len(decalages):
            i = 0
    return resultat
